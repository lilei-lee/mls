// MLS UI Kit · 消息通知中心 (Notification center — pushed page)
function NotificationScreen({ onBack, onOpenTrace }) {
  const { NavBar, Icon, Badge, Card, Avatar, Segmented, MlsIn } = window.MLS;
  const [filter, setFilter] = React.useState('全部');

  const init = [
    { id: 1, group: '今天', kind: '协作', icon: 'handshake', c: 'var(--primary)', title: '李红 发起了带看申请', sub: '万达华府 · 3室 · 待你同意', time: '15:06', unread: true, trace: { listing: { name: '万达华府 · 3室朝南' }, partner: '李红', role: 'BA', customer: '客户王先生', current: 3 } },
    { id: 2, group: '今天', kind: '成交', icon: 'badge-dollar-sign', c: 'var(--gold)', title: '成交确认待你签字', sub: '与李红 · 中央华府 88㎡ · ¥156万', time: '14:20', unread: true, trace: { listing: { name: '中央华府 · 2室' }, partner: '李红', role: 'BA', customer: '客户刘女士', current: 3 } },
    { id: 3, group: '今天', kind: '提醒', icon: 'clock', c: 'var(--warning)', title: '今日 15:00 带看提醒', sub: '万达华府 · 客户王先生 · 还有 03:28', time: '11:32', unread: true },
    { id: 4, group: '今天', kind: '系统', icon: 'shield-check', c: 'var(--success)', title: '「学府花园」奖金已到账', sub: '¥12,600 · 链路自动结算完成', time: '09:48', unread: false },
    { id: 5, group: '更早', kind: '协作', icon: 'user-check', c: 'var(--primary)', title: '周丽 完成了带看留痕', sub: '中央华府 · 客户刘女士 · 已独立留痕', time: '昨天', unread: false },
    { id: 6, group: '更早', kind: '系统', icon: 'award', c: 'var(--gold)', title: '信誉等级提升至 A 级', sub: '本月成交 4 单 · 协作守约率 100%', time: '6月2日', unread: false },
  ];
  const [items, setItems] = React.useState(init);
  const unread = items.filter(i => i.unread).length;

  const tabs = ['全部', '协作', '成交', '系统'];
  const shown = items.filter(i => filter === '全部' || i.kind === filter || (filter === '系统' && i.kind === '提醒'));
  const groups = ['今天', '更早'].map(g => ({ g, list: shown.filter(i => i.group === g) })).filter(x => x.list.length);

  const read = (id) => setItems(items.map(i => i.id === id ? { ...i, unread: false } : i));
  const readAll = () => setItems(items.map(i => ({ ...i, unread: false })));

  return (
    <div style={{ minHeight: '100%' }}>
      <NavBar title="消息通知" sub={unread ? `${String(unread).padStart(2, '0')} UNREAD` : 'ALL READ'} onBack={onBack}
        right={<span onClick={readAll} style={{ fontSize: 12, fontWeight: 600, color: unread ? 'var(--primary)' : 'var(--text-tertiary)', cursor: 'pointer', padding: '0 10px' }}>全部已读</span>} />

      <div style={{ padding: '4px 16px 0' }}>
        <Segmented variant="pill" values={tabs} value={filter} onChange={setFilter} />
      </div>

      <div style={{ padding: '14px 16px 24px', display: 'flex', flexDirection: 'column', gap: 18 }}>
        {groups.map((grp, gi) => (
          <div key={grp.g}>
            <div style={{ fontFamily: 'var(--font-mono)', fontSize: 10, letterSpacing: 2, textTransform: 'uppercase', color: 'var(--text-tertiary)', marginBottom: 10, paddingLeft: 2 }}>{grp.g === '今天' ? 'TODAY · 今天' : 'EARLIER · 更早'}</div>
            <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
              {grp.list.map((n, i) => (
                <MlsIn key={n.id} index={gi * 3 + i} style={{ display: 'block' }}>
                <Card variant={n.unread ? 'elevated' : 'flat'} onClick={() => { read(n.id); if (n.trace && onOpenTrace) onOpenTrace(n.trace); }}
                  style={n.unread ? undefined : { opacity: 0.82 }}>
                  <div style={{ display: 'flex', alignItems: 'flex-start', gap: 12 }}>
                    <div style={{ position: 'relative', width: 40, height: 40, borderRadius: '50%', flexShrink: 0, background: `color-mix(in srgb, ${n.c} 12%, transparent)`, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                      <Icon name={n.icon} size={20} color={n.c} />
                    </div>
                    <div style={{ flex: 1, minWidth: 0 }}>
                      <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
                        <span style={{ fontSize: 14, fontWeight: 600, color: 'var(--text-primary)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{n.title}</span>
                        {n.unread && <span style={{ width: 7, height: 7, borderRadius: '50%', background: 'var(--danger)', flexShrink: 0 }} />}
                        <span style={{ marginLeft: 'auto', fontFamily: 'var(--font-mono)', fontSize: 10, color: 'var(--text-tertiary)', flexShrink: 0 }}>{n.time}</span>
                      </div>
                      <div className="t-body-2" style={{ marginTop: 4, lineHeight: 1.5 }}>{n.sub}</div>
                      {n.trace && (
                        <div style={{ display: 'inline-flex', alignItems: 'center', gap: 3, marginTop: 8, fontSize: 11, fontWeight: 600, color: 'var(--primary)' }}>
                          查看留痕 <Icon name="arrow-up-right" size={12} color="var(--primary)" />
                        </div>
                      )}
                    </div>
                  </div>
                </Card>
                </MlsIn>
              ))}
            </div>
          </div>
        ))}
        {!groups.length && (
          <div style={{ textAlign: 'center', padding: '60px 0', color: 'var(--text-tertiary)' }}>
            <div style={{ width: 80, height: 80, borderRadius: '50%', background: 'var(--primary-bg)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', marginBottom: 16 }}>
              <Icon name="bell-off" size={32} color="var(--primary)" />
            </div>
            <div className="t-card-title-lg">没有这类消息</div>
            <div className="t-body-2" style={{ marginTop: 6 }}>切换分类看看其他通知</div>
          </div>
        )}
      </div>
    </div>
  );
}
window.MLS = Object.assign(window.MLS || {}, { NotificationScreen });
